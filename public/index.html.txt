<script>
    const socket = io();
    const feedContainer = document.getElementById('feed-container');
    const emptyMessage = document.getElementById('empty-message');

    // Triggered instantly whenever the backend detects a status shift
    socket.on('status-update', (data) => {
        if (emptyMessage) emptyMessage.remove();

        const card = document.createElement('div');
        card.className = `embed-card ${data.type}`;

        const isRestored = data.type === 'restored';
        const titleIcon = isRestored ? '✅' : '🚫';
        const titleText = isRestored ? 'Game Restored / Undeleted' : 'Game Deleted / Restricted';
        const actionText = isRestored ? 'is now active again and no longer marked as deleted.' : 'has been detected as deleted or restricted.';

        card.innerHTML = `
            <div class="embed-content">
                <div class="status-title">${titleIcon} ${titleText}</div>
                <div class="description">
                    ✒️ <a href="https://www.roblox.com/games/${data.placeId}" target="_blank" class="game-link">${data.name}</a> ${actionText}
                </div>
                <div class="field">
                    <span class="field-label">Place ID</span>
                    <span class="field-value">${data.placeId}</span>
                </div>
                <div class="embed-footer">
                    Deletion Monitor • ${data.time}
                </div>
            </div>
        `;

        // Prepend so the newest ban alerts show up at the top
        feedContainer.insertBefore(card, feedContainer.firstChild);
    });
</script>
